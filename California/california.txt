Let's try a bbox with overlay

from shapely.geometry import Polygon
xmin, ymin, xmax, ymax = -120, -117.5, 33, 35
poly = Polygon([(xmin,ymin),(xmax,ymin),(xmax,ymax),(xmin,ymax)])
gs = gpd.GeoSeries(poly)
bbox = gpd.GeoDataFrame({'geometry': gs})
bbox = bbox.set_crs(calif.crs)

socal = calif.overlay(bbox, how='intersection')
socal.boundary.plot()
plt.show()
