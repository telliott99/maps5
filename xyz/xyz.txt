the coverage for a tile depends on LAT, but not so for LON

pseudocode
n = 2 ^ zoom
xtile = n * ((lon_deg + 180) / 360)
ytile = n * (1 - (log(tan(lat_rad) + sec(lat_rad)) / π)) / 2

log = ln(tan(lat_rad) + sec(lat_rad))
delta = 1 - log/pi
ytile = delta * 2^{z-1}

(where log is ln)

n = 2 ^ zoom
lon_deg = xtile / n * 360.0 - 180.0
lat_rad = arctan(sinh(π * (1 - 2 * ytile / n)))
lat_deg = lat_rad * 180.0 / π

convert to Python

from math import *
z = 13
n = 2**z

def f(LAT):
    rad = radians(LAT)
    s = tan(rad) + 1/cos(rad)
    d = 1 - log(s)/pi
    return d*2**(z-1)
